import {createApp,h,ref,markRaw} from 'vue';
import ElementPlus from 'element-plus';
import 'element-plus/dist/index.css';
import '@flowduet/designer/style.css';
import '@vue-flow/core/dist/style.css';
import '@flowduet/form-create/style.css';
import {BpmnModel,flowableAdapter,CORE_VERSION} from '@flowduet/core';
import {DingtalkDesigner,BpmnCanvas} from '@flowduet/designer';
import {FlowDesignSession,FormDesigner,FormPreview,exportDeployXml} from '@flowduet/form-create';
const seed=BpmnModel.create({processId:'seed',adapter:flowableAdapter});const session=new FlowDesignSession(seed);
const input=await(await fetch('/design.flowduet.json')).text();let state=await session.open(input);
createApp({setup(){const form=ref(markRaw(state.forms[0])),status=ref('已从实际tarball打开两表单文档'),design=ref(false);const model=ref(markRaw(state.model));
return()=>h('main',{style:'padding:20px;max-width:1000px;margin:auto'},[
h('h1','三包版本化外部消费验收'),h('p','core '+CORE_VERSION+' · designer 0.1.0 · form-create 0.1.0'),h('p',{'data-test':'consumer-status'},status.value),
h('button',{onClick:()=>{design.value=!design.value}},design.value?'返回预览':'打开真实设计器'),
h('button',{onClick:async()=>{const saved=await session.save();const opened=await session.open(JSON.stringify(saved.document));form.value=markRaw(opened.forms[0]);model.value=markRaw(opened.model);status.value='保存恢复通过：'+opened.forms.length+'张表单';}},'保存并恢复'),
h('button',{onClick:async()=>{const xml=await exportDeployXml(session.current.model,session.current.forms);const link=document.createElement('a');link.href=URL.createObjectURL(new Blob([xml]));link.download='release-consumer.xml';link.click();status.value='组合导出已下载';}},'下载组合XML'),
design.value?h(FormDesigner,{name:form.value.name,rules:form.value.rules,options:form.value.options,onSave:(rules,options)=>{session.updateFormContent(form.value.id,rules,options);form.value=markRaw(session.current.forms[0]);design.value=false;status.value='真实设计器成对保存通过';}}):h(FormPreview,{form:form.value}),
h('details',[h('summary','公开流程组件'),h(DingtalkDesigner,{model:model.value,formOptions:session.current.forms.map(f=>({id:f.id,name:f.name}))}),h(BpmnCanvas,{model:model.value})])]);}}).use(ElementPlus).mount('#app');
