import { mkdir, writeFile } from "node:fs/promises";
import { resolve } from "node:path";
import { pathToFileURL } from "node:url";

const [driverPath, baseUrl, outputDir, outputName = "style-probe.json"] = process.argv.slice(2);
if (!driverPath || !baseUrl || !outputDir) throw new Error("需要驱动、URL 和输出目录参数");
const { chromium } = await import(pathToFileURL(resolve(driverPath)).href);
await mkdir(outputDir, { recursive: true });
const browser = await chromium.launch({
  headless: true,
  executablePath:
    process.env.FLOWDUET_CHROMIUM_PATH ||
    "/Users/l/Library/Caches/ms-playwright/chromium-1248/chrome-mac-x64/Google Chrome for Testing.app/Contents/MacOS/Google Chrome for Testing",
});
const page = await browser.newPage({ viewport: { width: 1440, height: 2200 }, deviceScaleFactor: 1 });
const errors = [];
const failures = [];
page.on("pageerror", (error) => errors.push(`pageerror: ${error.stack || error.message}`));
page.on("console", (message) => {
  if (message.type() === "error") errors.push(`console: ${message.text()}`);
});
const check = (condition, message) => {
  if (!condition) failures.push(message);
};
try {
  await page.route("**/favicon.ico", (route) => route.fulfill({ status: 204, body: "" }));
  await page.goto(baseUrl, { waitUntil: "networkidle" });
  await page.locator('[data-test="form-appearance-scene-btn"]').click();
  await page.locator('[data-test="form-appearance-first-controls"]').click({ timeout: 2000 });
  const preview = page.locator(
    '[data-test="form-appearance-first-instance"] [data-test="form-preview"]',
  );
  await preview.waitFor({ state: "visible" });
  await page.evaluate(() => {
    const source = document.querySelector('[data-test="form-appearance-first-source"]');
    source.value = "brand";
    source.dispatchEvent(new Event("change", { bubbles: true }));
  });
  await page.waitForTimeout(200);
  await preview.locator(".el-select__wrapper").first().click();
  const panel = page.locator(".el-select__popper.preview-select-panel");
  await panel.waitFor({ state: "visible" });
  await panel.locator(".el-select-dropdown__item").filter({ hasText: "同意" }).first().hover();
  await page.waitForTimeout(200);
  const actual = await preview.evaluate((root) => {
    const select = root.querySelectorAll(".el-select")[0];
    const wrapper = select.querySelector(".el-select__wrapper");
    const caret = select.querySelector(".el-select__caret");
    const panel = document.querySelector(".el-select__popper.preview-select-panel");
    const items = [...panel.querySelectorAll(".el-select-dropdown__item")];
    const selected = items.find((item) => item.textContent === "拒绝");
    const hovering = items.find((item) => item.textContent === "同意");
    const disabled = items.find((item) => item.textContent === "禁用选项");
    const rect = (element) => {
      const box = element.getBoundingClientRect();
      return { width: box.width, height: box.height };
    };
    const style = (element) => {
      const value = getComputedStyle(element);
      return {
        color: value.color,
        background: value.backgroundColor,
        outlineWidth: value.outlineWidth,
        outlineOffset: value.outlineOffset,
        fontSize: value.fontSize,
        lineHeight: value.lineHeight,
        whiteSpace: value.whiteSpace,
      };
    };
    return {
      trigger: { ...rect(wrapper), ...style(wrapper), focused: document.activeElement === select.querySelector("input") },
      caret: style(caret),
      panel: style(panel),
      selected: { ...rect(selected), ...style(selected) },
      hovering: { ...rect(hovering), ...style(hovering) },
      disabled: style(disabled),
      ownerClasses: [...panel.classList].filter((name) => name.startsWith("flowduet-form-preview-select-")),
      panelVariables: panel.getAttribute("style"),
    };
  });
  check(actual.trigger.height === 32, `选择框高度应为 32，实际 ${actual.trigger.height}`);
  check(actual.trigger.focused, "打开面板后选择框输入未保持焦点");
  check(actual.caret.color === "rgb(32, 38, 61)", `caret 未消费正文色：${actual.caret.color}`);
  check(actual.trigger.outlineWidth === "2px", `焦点环宽度应为 2px，实际 ${actual.trigger.outlineWidth}`);
  check(actual.trigger.outlineOffset === "3px", `焦点环偏移应为 3px，实际 ${actual.trigger.outlineOffset}`);
  check(actual.panel.background === "rgb(255, 255, 255)", `面板表面色错误：${actual.panel.background}`);
  check(actual.selected.color === "rgb(45, 62, 151)", `选中前景错误：${actual.selected.color}`);
  check(actual.selected.background === "rgb(234, 240, 255)", `选中背景错误：${actual.selected.background}`);
  check(actual.selected.height === 32, `舒适选项行高应为 32，实际 ${actual.selected.height}`);
  check(actual.hovering.background === "rgb(240, 242, 248)", `键盘高亮背景错误：${actual.hovering.background}`);
  check(actual.disabled.color === "rgb(139, 146, 164)", `禁用选项前景错误：${actual.disabled.color}`);
  check(actual.ownerClasses.length === 1, `实例类不完整：${actual.ownerClasses}`);
  const result = { browser: await browser.version(), actual, errors, failures };
  await writeFile(resolve(outputDir, outputName), JSON.stringify(result, null, 2));
  await page.screenshot({ path: resolve(outputDir, outputName.replace(/\.json$/, ".png")), fullPage: true });
  console.log(JSON.stringify(result, null, 2));
  if (failures.length > 0 || errors.length > 0) process.exitCode = 1;
} catch (error) {
  console.error(await page.locator("body").innerText());
  console.error(JSON.stringify(errors));
  await page.screenshot({ path: resolve(outputDir, "probe-error.png"), fullPage: true });
  throw error;
} finally {
  await browser.close();
}
