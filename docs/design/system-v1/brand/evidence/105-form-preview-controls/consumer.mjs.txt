import { writeFile } from "node:fs/promises";
import { resolve } from "node:path";
import { pathToFileURL } from "node:url";

const [driverPath, baseUrl, outputFile] = process.argv.slice(2);
if (!driverPath || !baseUrl || !outputFile) throw new Error("需要驱动、URL 和输出文件参数");
const { chromium } = await import(pathToFileURL(resolve(driverPath)).href);
const browser = await chromium.launch({
  headless: true,
  executablePath:
    process.env.FLOWDUET_CHROMIUM_PATH ||
    "/Users/l/Library/Caches/ms-playwright/chromium-1248/chrome-mac-x64/Google Chrome for Testing.app/Contents/MacOS/Google Chrome for Testing",
});
const page = await browser.newPage({ viewport: { width: 1100, height: 1200 } });
const errors = [];
const failures = [];
page.on("pageerror", (error) => errors.push(error.message));
page.on("console", (message) => {
  if (message.type() === "error") errors.push(message.text());
});
const check = (condition, message) => {
  if (!condition) failures.push(message);
};
try {
  await page.goto(baseUrl, { waitUntil: "networkidle" });
  const preview = page.locator('[data-test="form-preview"]');
  await preview.waitFor({ state: "visible" });
  await page.waitForTimeout(350);
  const actual = await preview.evaluate((root) => {
    const number = root.querySelector(".el-input-number");
    const radio = root.querySelectorAll(".el-radio")[1];
    const radioInner = radio.querySelector(".el-radio__inner");
    const selectedBox = root.querySelectorAll(".el-checkbox")[0];
    const boxInner = selectedBox.querySelector(".el-checkbox__inner");
    const mixedInner = root.querySelectorAll(".el-checkbox")[1].querySelector(".el-checkbox__inner");
    const switchRoot = root.querySelector(".el-switch");
    const switchCore = switchRoot.querySelector(".el-switch__core");
    const switchAction = switchRoot.querySelector(".el-switch__action");
    return {
      className: root.className,
      primary: root.style.getPropertyValue("--el-color-primary"),
      background: getComputedStyle(root).backgroundColor,
      numberHeight: number.getBoundingClientRect().height,
      numberControlBorder: getComputedStyle(number.querySelector(".el-input-number__decrease")).borderRightColor,
      radioBackground: getComputedStyle(radioInner).backgroundColor,
      radioDot: getComputedStyle(radioInner, "::after").backgroundColor,
      checkboxBackground: getComputedStyle(boxInner).backgroundColor,
      checkboxCheck: getComputedStyle(boxInner, "::after").borderRightColor,
      mixed: root.querySelectorAll(".el-checkbox")[1].querySelector("input").indeterminate,
      mixedBar: getComputedStyle(mixedInner, "::before").backgroundColor,
      switchTrack: getComputedStyle(switchCore).backgroundColor,
      switchAction: getComputedStyle(switchAction).backgroundColor,
      buttonHeight: root.querySelector('[data-test="form-preview-validate"]').getBoundingClientRect().height,
      formClass: root.querySelector(".el-form").className,
    };
  });
  check(actual.className.includes("flowduet-appearance--brand"), "公开 appearance 未生效");
  check(actual.primary === "#A5BBFF" && actual.background === "rgb(29, 36, 53)", "公开品牌深色变量或 CSS 缺失");
  check(actual.numberHeight === 24 && actual.numberControlBorder === "rgb(117, 131, 159)", "公开数字密度或边界未生效");
  check(actual.radioBackground === "rgb(165, 187, 255)" && actual.radioDot === "rgb(23, 31, 62)", "公开单选选中配对未生效");
  check(actual.checkboxBackground === "rgb(165, 187, 255)" && actual.checkboxCheck === "rgb(23, 31, 62)", "公开多选选中配对未生效");
  check(actual.mixed && actual.mixedBar === "rgb(23, 31, 62)", "公开半选标记未生效");
  check(actual.switchTrack === "rgb(117, 131, 159)" && actual.switchAction === "rgb(29, 36, 53)", "公开开关关闭配对未生效");
  check(actual.buttonHeight === 24 && actual.formClass.includes("el-form--label-top"), "公开密度或缺省标签适配缺失");
  await preview.locator(".el-input-number__increase").click();
  await preview.locator(".el-radio").nth(0).locator(".el-radio__inner").click();
  await preview.locator(".el-checkbox").nth(1).locator(".el-checkbox__inner").click();
  await preview.locator(".el-switch__core").click();
  await preview.locator("input").first().fill("实际安装包文本");
  await preview.locator("textarea").fill("实际安装包多行试填");
  await preview.locator('[data-test="form-preview-validate"]').click();
  await preview.locator('[data-test="form-preview-validation"]').filter({ hasText: "校验通过" }).waitFor();
  await page.waitForTimeout(350);
  const values = JSON.parse(await preview.locator('[data-test="form-preview-values"]').textContent());
  check(values.amount === 2 && values.decision === "yes" && values.tags.includes("mixed") && values.notify === true, `公开包四类试填值错误 ${JSON.stringify(values)}`);
  const switchOn = await preview.locator(".el-switch").evaluate((element) => ({
    track: getComputedStyle(element.querySelector(".el-switch__core")).backgroundColor,
    action: getComputedStyle(element.querySelector(".el-switch__action")).backgroundColor,
  }));
  check(switchOn.track === "rgb(165, 187, 255)" && switchOn.action === "rgb(23, 31, 62)", "公开开关开启配对未生效");
  const resources = await page.evaluate(() => performance.getEntriesByType("resource").map((entry) => entry.name));
  check(!resources.some((url) => url.includes("/Users/l/Documents/projects/flowduet/packages/")), "消费工程使用源码 alias");
  check(errors.length === 0, `浏览器错误：${errors.join("；")}`);
  await preview.screenshot({ path: resolve(outputFile, "..", "consumer.png") });
  const summary = { actual, values, switchOn, resources, errors, failures, result: failures.length === 0 ? "公开 JS / 严格 TS / CSS / 四类控件浏览器渲染通过" : "失败" };
  await writeFile(outputFile, JSON.stringify(summary, null, 2));
  console.log(JSON.stringify({ errors, failures, values, switchOn }, null, 2));
  if (failures.length > 0) process.exitCode = 1;
} finally {
  await browser.close();
}
