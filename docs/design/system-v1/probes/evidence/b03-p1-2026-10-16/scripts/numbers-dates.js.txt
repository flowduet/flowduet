async page => {
  await page.getByRole('button',{name:'B03 P1 修复'}).click();
 const rows=[];
 const root=i=>page.locator(`[data-repair-instance="${i}"]`);
 const action=async selector=>{const out=page.locator('[data-test="repair-result"]');const before=await out.getAttribute('data-sequence');await page.locator(selector).click();await page.waitForFunction(old=>document.querySelector('[data-test="repair-result"]').getAttribute('data-sequence')!==old,before);return JSON.parse(await out.textContent());};
 const select=async(i,field)=>{await root(i).locator('._fc-m-drag ._fd-drag-mask').nth(field==='number'?2:6).click(); const input=root(i).locator(`[data-retained-input="${field}"]`);await input.waitFor({state:'visible'});return input;};
 const find=(rules,field)=>{for(const r of rules){if(r.field===field)return r;const c=find(r.children||[],field);if(c)return c;}};
 const assert=(ok,detail)=>{if(!ok)throw Error(JSON.stringify(detail));};
 for(const layout of ['top','grid']) {
  await page.locator('[data-test="repair-layout"]').selectOption(layout);
  await page.locator('[data-test="repair-reload"]').click();
  for(let round=1;round<=3;round++) {
   for(const [kind,text] of [['number','-'],['number','1.'],['number','1e'],['date','2026-'],['date','2026-02-30']]) {
    const input=await select(0,kind);await input.fill(text);
    await page.locator('[data-test="repair-observe"]').click();const revisionBefore=JSON.parse(await page.locator('[data-test="repair-result"]').textContent()).states[0].revision;
    await page.locator('[data-test="repair-nav-1"]').click();await page.locator('[data-test="repair-nav-0"]').click();
    await page.locator('[data-test="repair-phase"]').click();await page.locator('[data-test="repair-phase"]').click();
    const after=await input.inputValue();const o=await action('[data-test="repair-observe"]');
    const captured=o.captures[0]; assert(o.states[0].revision===revisionBefore&&after===text&&o.states[0].canSuspend&&!o.states[0].canCapture&&captured.status==='blocked'&&o.liveUnchanged&&o.persistCalls===0&&o.adaptersAndLeasesCurrent,{after,o});
    rows.push({id:'N01/N04',layout,round,kind,text,after,state:o.states[0],liveUnchanged:o.liveUnchanged,pass:true});
    await input.fill(kind==='number'?'0':'2024-02-29');
   }
   for(const [text,value] of [['0',0],['-2',-2],['1.5',1.5],['.5',.5],['1e3',1000]]) {
    const input=await select(0,'number');await input.fill(text);
    await page.waitForFunction(()=>JSON.parse(document.querySelector('[data-test="repair-states"]').textContent)[0]?.canCapture===true);
    await page.locator('[data-test="repair-nav-1"]').click();await page.locator('[data-test="repair-nav-0"]').click();
    const o=await action('[data-test="repair-observe"]');const output=o.captures[0];assert(output.status==='captured',{text,o});
    const actual=find(JSON.parse(output.content.rules),'a_amount').value;assert(actual===value&&(await input.inputValue())===text,{text,actual});
    rows.push({id:'N02',layout,round,text,value:actual,raw:await input.inputValue(),pass:true});
   }
  }
  for(const text of ['', ' 1','NaN','Infinity']) {
   const input=await select(0,'number');await input.fill(text);const o=await action('[data-test="repair-observe"]');
   if(text===''){assert(o.captures[0].status==='captured',o);assert(!Object.hasOwn(find(JSON.parse(o.captures[0].content.rules),'a_amount'),'value'),o);}
   else assert(o.captures[0].status==='blocked'&&(await input.inputValue())===text,o);
   rows.push({id:'N03',layout,text,status:o.captures[0].status,pass:true});await input.fill('0');
  }
 }
 return {viewport:page.viewportSize(),dpr:await page.evaluate(()=>devicePixelRatio),source:'real Chromium UI; no OS IME claim',rows};
}
