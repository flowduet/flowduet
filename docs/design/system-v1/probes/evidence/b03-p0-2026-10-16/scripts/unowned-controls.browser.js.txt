async page => {
 page.setDefaultTimeout(7000);const rows=[];const root=page.locator('[data-seam-owner=A]');
 const events=async()=>JSON.parse(await page.locator('[data-test=seam-events]').textContent());
 for(const layout of ['top','grid']){
  await page.locator('[data-test=seam-layout]').selectOption(layout);await page.locator('[data-test=seam-load]').click();
  await root.locator('._fc-m-drag ._fd-drag-tool').filter({has:page.locator('.el-form-item__label',{hasText:'甲单选'})}).last().click();

  await page.locator('[data-test=seam-clear]').click();await input.fill('');await input.press('-');

  await page.locator('[data-test=seam-clone-A]').evaluate(el=>el.click());const trace=await events();

 }
 return rows;
}
