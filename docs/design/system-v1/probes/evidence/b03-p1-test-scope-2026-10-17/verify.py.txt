import subprocess,json,time,pathlib,hashlib,sys
out=pathlib.Path('/tmp/b03-test-scope-repair-2026-10-17')
base=['pnpm','--filter','@flowduet/playground','exec','vitest','run']
files=['src/App.integration.test.ts','probes/stage-zero/location.test.ts']
target='插入抄送草稿后自动与手动导出|新建草稿可保存并报告待修复项|读取期间新打开表单编辑器|单↔多转换后原分支诊断'
commands=[('scope-green',base+['src/test-utils/browser-test-scope.test.ts'])]+[(f'target-{i}',base+files+['-t',target]) for i in range(1,4)]+[('app-file',base+[files[0]]),('location-file',base+[files[1]]),('full-default',['pnpm','test']),('lint',['pnpm','lint']),('build',['pnpm','build']),('strict',['pnpm','--filter','@flowduet/form-create','exec','vue-tsc','-p','../../apps/playground/probes/stage-zero/tsconfig.json','--noEmit']),('esm',['node','--input-type=module','-e','import { CORE_VERSION } from "./packages/core/dist/index.js"; console.log("dist import OK",CORE_VERSION)'])]
records=json.loads((out/"commands.json").read_text()) if len(sys.argv)>1 else []

for name,cmd in commands:

 with (out/(name+'.log')).open('w') as log:r=subprocess.run(cmd,stdout=log,stderr=subprocess.STDOUT)
 records.append(dict(name=name,command=cmd,exitCode=r.returncode,seconds=round(time.time()-start,2)))
