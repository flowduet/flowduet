   rows.push({case:'busy-preblur-'+control,run,source:'native-pointer-keyboard',stillFocused,trace,pass:await visible()&&stillFocused&&await raw.inputValue()==='busy-原文'&&trace.some(e=>e.event==='dialog-preflight'&&e.detail.active==='dialog-raw')&&!trace.some(e=>e.event==='dialog-blur')});

  }

 }

 await raw.dispatchEvent('compositionend',{data:'中'});await dismiss();

}
