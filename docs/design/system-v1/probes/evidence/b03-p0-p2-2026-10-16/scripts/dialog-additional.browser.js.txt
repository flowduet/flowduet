async page => {
 page.setDefaultTimeout(7000); const rows=[];
 const events=async()=>JSON.parse(await page.locator('[data-test=seam-events]').textContent());
 const clear=()=>page.locator('[data-test=seam-clear]').click();
 await page.locator('[data-test=dialog-guarded]').check();await page.locator('[data-test=dialog-busy]').check();
 await clear();await page.locator('[data-test=dialog-open]').click();
 const raw=page.locator('[data-test=dialog-raw]');await raw.fill('拒绝后不可借click关闭');
 const x=page.locator('[data-test=dialog-x]');const box=await x.boundingBox();
 await page.mouse.move(box.x+box.width/2,box.y+box.height/2);await page.mouse.down();
 await page.locator('[data-test=dialog-busy]').evaluate(el=>el.click());await page.mouse.up();
 let trace=await events();rows.push({case:'refused-pointerdown-late-click',source:'native-pointer-plus-owned-busy-protocol',trace,pass:await raw.isVisible()&&await raw.inputValue()==='拒绝后不可借click关闭'&&trace.some(e=>e.event==='dialog-before-close'&&e.detail.refusedGesture===true)});
 await page.locator('[data-test=dialog-program-close]').evaluate(el=>el.click());await page.locator('.b03-seam-dialog').waitFor({state:'hidden'});
 await page.locator('[data-test=dialog-busy]').check();await clear();await page.locator('[data-test=dialog-open]').click();
 await x.focus();await x.press('Enter');trace=await events();rows.push({case:'busy-owned-key-activation',source:'native-keyboard',trace,pass:await raw.isVisible()&&trace.some(e=>e.event==='dialog-preflight'&&e.detail.key==='Enter')});
 await page.locator('[data-test=dialog-program-close]').evaluate(el=>el.click());await page.locator('.b03-seam-dialog').waitFor({state:'hidden'});
 await page.locator('[data-test=dialog-busy]').uncheck();await page.locator('[data-test=dialog-guarded]').uncheck();
 for(const control of ['dialog-x','mask']) {
  await clear();await page.locator('[data-test=dialog-open]').click();await raw.fill('default-close');
  if(control==='mask')await page.locator('.b03-seam-modal').click({position:{x:5,y:5}});else await x.click();
  await page.locator('.b03-seam-dialog').waitFor({state:'hidden'});trace=await events();rows.push({case:'default-'+control+'-close',source:'native-pointer',trace,pass:trace.some(e=>e.event==='dialog-before-close')});
 }
 await clear();await page.locator('[data-test=seam-number-B]').click();const foreign=page.locator('[data-seam-input=B] [data-test=seam-raw]');await foreign.fill('foreign');await foreign.press('Escape');
 trace=await events();rows.push({case:'other-instance-native-input-unaffected',source:'native-pointer-keyboard',trace,pass:await foreign.inputValue()==='foreign'&&!trace.some(e=>e.event==='dialog-preflight')&&!trace.some(e=>e.event==='dialog-before-close')});
 return rows;
}
