import { afterEach, describe, expect, it, vi } from "vitest";
import { flushPromises, mount } from "@vue/test-utils";
import FormDesigner from "../../../../packages/form-create/src/components/FormDesigner.vue";
import { describeValue, equivalentContent, observeLiveState } from "./form-state/state-evidence";

const mounted: { unmount(): void }[] = [];
afterEach(() => {
  try { mounted.splice(0).forEach((view) => view.unmount()); }
  finally { document.body.innerHTML = ""; }
});
const content = (rule: Record<string, unknown>) => ({ rules: JSON.stringify([rule]), options: "{}" });
describe("C0 表单状态观察", () => {
  it.each([
    [{}, { present: false, type: "undefined", json: null }],
    [{ value: 0 }, { present: true, type: "number", json: "0" }],
    [{ value: false }, { present: true, type: "boolean", json: "false" }],
    [{ value: [] }, { present: true, type: "array", json: "[]" }],
    [{ value: "" }, { present: true, type: "string", json: '\"\"' }],
    [{ value: null }, { present: true, type: "object", json: "null" }],
  ])("value_presence_and_types_are_distinct：%j", (rule, expected) => {
    expect(describeValue(rule)).toEqual(expected);
  });
  it.each([[0, "0"], [false, undefined], [[], undefined], ["", undefined]])(
    "不把不同默认值归一：%j / %j", (left, right) => {
      const base = { type: "input", field: "a" };
      expect(equivalentContent(content({ ...base, value: left }), content({ ...base, value: right }))).toBe(false);
    },
  );
  it("允许运行态元数据变化，但字段/选项/列顺序变化不等价", () => {
    const a = { rules: '[{"type":"select","field":"a","options":[{"label":"甲","value":0},{"label":"乙","value":"0"}]}]', options: '{"a":1,"b":2}' };
    expect(equivalentContent(a, { ...a, rules: a.rules.replace('"type"', '"_fc_id":"new","type"'), options: '{"b":2,"a":1}' })).toBe(true);
    expect(equivalentContent(a, { ...a, rules: a.rules.replace('"field":"a"', '"field":"b"') })).toBe(false);
    expect(equivalentContent(a, { ...a, rules: a.rules.replace('[{"label":"甲","value":0},{"label":"乙","value":"0"}]', '[{"label":"乙","value":"0"},{"label":"甲","value":0}]') })).toBe(false);
    const grid = { rules: '[{"type":"fcRow","children":[{"type":"col","children":[{"type":"input","field":"a","value":[]}]},{"type":"col","children":[]}]}]', options: '{}' };
    expect(equivalentContent(grid, { ...grid, rules: grid.rules.replace(',"value":[]', '') })).toBe(false);
    expect(equivalentContent(a, { ...a, rules: "bad" })).toBe(false);
  });
  it("observe_does_not_flush_focused_input：真实设计器菜单、焦点文字及选区保留", async () => {
    const view = mount(FormDesigner, { props: { name: "C观察", rules: '[{"type":"input","field":"a_subject","title":"事项甲","value":"旧值"}]', options: "{}" }, attachTo: document.body });
    mounted.push(view);
    await flushPromises();
    expect(view.find("._fc-l-item").exists()).toBe(true);
    await view.find("._fc-m-drag .el-form-item").trigger("click");
    await flushPromises();
    await vi.waitFor(() => expect(view.findAll("._fc-r-config label").map((el) => el.text())).toContain("默认值"));
    const row = view.findAll("._fc-r-config .el-form-item").find((item) => item.find("label").exists() && item.find("label").text() === "默认值");
    expect(row).toBeTruthy();
    const input = row!.find("input");
    input.element.focus();
    await input.setValue("中文末尾未失焦");
    input.element.setSelectionRange(2, 5);
    const observed = observeLiveState(view.element as HTMLElement, { composing: false, dragging: false });
    expect(observed.pendingControls.some((item) => item.text === "中文末尾未失焦" && item.selection?.[0] === 2 && item.selection[1] === 5)).toBe(true);
    expect(document.activeElement).toBe(input.element);
    expect(input.element.value).toBe("中文末尾未失焦");
    expect(view.emitted("save")).toBeUndefined();
    expect(view.props("rules")).toContain("旧值");
  });
});
