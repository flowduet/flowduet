def run(cmd,name,timeout=30,capture=False):

  except subprocess.TimeoutExpired:

   try:p.communicate(timeout=15)
   except subprocess.TimeoutExpired:os.killpg(p.pid,signal.SIGKILL);p.communicate()
   records['commands'].append({'name':name,'command':cmd,'timeout':timeout,'exit':'timeout'});save();raise
  if capture:log.write(stdout)
 records['commands'].append({'name':name,'command':cmd,'exit':p.returncode});save()
