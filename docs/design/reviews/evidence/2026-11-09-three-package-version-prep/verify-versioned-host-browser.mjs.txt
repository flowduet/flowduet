async function drop(selector) {return page.locator(selector).evaluate(el => {
 const event=new DragEvent("drop", {bubbles:true,cancelable:true}); el.dispatchEvent(event); return {defaultPrevented:event.defaultPrevented};

async function unloadSignal() {return page.evaluate(() => {

 window.dispatchEvent(event); return event.returnValue;
});}
